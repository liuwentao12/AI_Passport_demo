cmake_minimum_required(VERSION 3.22)
project(identity_ui_tests C CXX)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../managed_components/lvgl__lvgl lvgl)
add_executable(identity_test identity_test.c ../main/app/identity.c)
target_include_directories(identity_test PRIVATE ../main)
add_test(NAME identity_validation COMMAND identity_test)
add_executable(ui_test ui_test.c ../main/app/identity.c ../main/board/touch.c ../main/ui/ui.c
    ../main/ui/ui_theme.c ../main/ui/pages/identity_page.c ../main/ui/pages/details_page.c)
target_include_directories(ui_test PRIVATE ../main)
target_link_libraries(ui_test PRIVATE lvgl)
add_test(NAME ui_layout_and_navigation COMMAND ui_test)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(identity_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_compile_options(ui_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
